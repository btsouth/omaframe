# GuiPrivate is used for one QPA header (the window's Wayland surface); no
# private symbol is linked, so the version warning does not apply.

foreach(PROTOCOL ext-foreign-toplevel-list ext-image-capture-source ext-image-copy-capture)
  set(XML /usr/share/wayland-protocols/staging/${PROTOCOL}/${PROTOCOL}-v1.xml)
  set(HEADER ${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-v1-client-protocol.h)
  set(CODE ${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-v1-protocol.c)
  add_custom_command(OUTPUT ${HEADER} ${CODE}
    COMMAND ${WAYLAND_SCANNER} client-header ${XML} ${HEADER}
    COMMAND ${WAYLAND_SCANNER} private-code ${XML} ${CODE} DEPENDS ${XML} VERBATIM)
  list(APPEND PROTOCOL_SOURCES ${HEADER} ${CODE})
endforeach()

list(APPEND PROTOCOL_SOURCES ${HEADER} ${CODE})
